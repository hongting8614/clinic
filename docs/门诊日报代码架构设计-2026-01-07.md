## 一、当前架构（已实现）✅

### 1.1 核心逻辑

```javascript
formatDailyReport(records, dateStr, locationName) {
  // 1. 聚合统计数据
  const stats = this.aggregateStats(records);
  
  // 2. 格式化日期
  const dateFormatted = this.formatDate(dateStr);
  
  // 3. 生成报告文本
  let report = '';
  
  // 3.1 总人数（只在有记录时显示）
  if (stats.total > 0) {
    report = `${dateFormatted}北京欢乐谷医务室（${locationName}）当日接诊${stats.total}人。\n`;
  }
  
  // 3.2 游客统计（只在有游客时显示）
  if (stats.visitor.length > 0) {
    report += this.formatVisitorStats(stats.visitor);
  }
  
  // 3.3 员工统计（只在有员工时显示）
  if (stats.employee.length > 0) {
    report += this.formatEmployeeStats(stats.employee);
  }
  
  // 3.4 出诊统计（只在有出诊时显示）
  if (stats.outcall.length > 0) {
    report += this.formatOutcallStats(stats.outcall);
  }
  
  return report.trim();
}
```

### 1.2 显示规则

| 条件 | 显示内容 | 示例 |
|------|---------|------|
| `stats.total = 0` | 不显示任何内容 | （空白） |
| `stats.total > 0` | 显示总人数 | `2026年1月7日北京欢乐谷医务室（陆园）当日接诊2人。` |

## 二、优化架构（推荐）⭐⭐⭐⭐⭐

### 2.1 设计原则

1. **单一职责**：每个方法只做一件事
2. **可测试性**：每个部分都可以独立测试
3. **可维护性**：代码结构清晰，易于修改
4. **可扩展性**：方便添加新的统计类型

### 2.2 优化后的代码结构

```javascript
// ==================== 主方法 ====================
formatDailyReport(records, dateStr, locationName) {
  const stats = this.aggregateStats(records);
  const dateFormatted = this.formatDate(dateStr);
  
  // 使用数组收集各部分内容，最后统一拼接
  const reportParts = [];
  
  // 添加标题（只在有记录时）
  const header = this.buildReportHeader(dateFormatted, locationName, stats.total);
  if (header) reportParts.push(header);
  
  // 添加游客统计
  const visitorSection = this.buildVisitorSection(stats.visitor);
  if (visitorSection) reportParts.push(visitorSection);
  
  // 添加员工统计
  const employeeSection = this.buildEmployeeSection(stats.employee);
  if (employeeSection) reportParts.push(employeeSection);
  
  // 添加出诊统计
  const outcallSection = this.buildOutcallSection(stats.outcall);
  if (outcallSection) reportParts.push(outcallSection);
  
  return reportParts.join('\n');
}

// ==================== 构建各部分内容 ====================

// 构建报告标题
buildReportHeader(dateFormatted, locationName, total) {
  if (total === 0) return '';
  return `${dateFormatted}北京欢乐谷医务室（${locationName}）当日接诊${total}人。`;
}

// 构建游客统计部分
buildVisitorSection(visitorStats) {
  if (!visitorStats || visitorStats.length === 0) return '';
  
  const visitorTotal = visitorStats.reduce((sum, item) => sum + item.total, 0);
  const visitorParts = visitorStats.map(item => {
    if (item.locations && item.locations.length > 0) {
      const locationParts = item.locations.map(loc => `${loc.name}${loc.count}人`);
      return `${item.disease}${item.total}人（${locationParts.join('，')}）`;
    }
    return `${item.disease}${item.total}人`;
  });
  
  return `游客${visitorTotal}人：${visitorParts.join('，')}。`;
}

// 构建员工统计部分
buildEmployeeSection(employeeStats) {
  if (!employeeStats || employeeStats.length === 0) return '';
  
  const employeeTotal = employeeStats.reduce((sum, item) => sum + item.total, 0);
  const employeeParts = employeeStats.map(item => `${item.disease}${item.total}人`);
  
  return `员工${employeeTotal}人：${employeeParts.join('，')}。`;
}

// 构建出诊统计部分
buildOutcallSection(outcallStats) {
  if (!outcallStats || outcallStats.length === 0) return '';
  
  const outcallTotal = outcallStats.reduce((sum, item) => sum + item.count, 0);
  const outcallParts = outcallStats.map(item => `${item.location}${item.count}次`);
  
  return `出诊${outcallTotal}次：${outcallParts.join('，')}。`;
}
```

## 三、完整代码实现

### 3.1 优化后的完整代码

```javascript
// ==================== 格式化日报 ====================
formatDailyReport(records, dateStr, locationName) {
  console.log('[formatDailyReport] 开始生成日报', {
    recordsCount: records.length,
    dateStr,
    locationName
  });
  
  // 1. 聚合统计数据
  const stats = this.aggregateStats(records);
  console.log('[formatDailyReport] 统计数据:', stats);
  
  // 2. 格式化日期
  const dateFormatted = this.formatDate(dateStr);
  
  // 3. 构建报告各部分
  const reportParts = [];
  
  // 3.1 标题部分
  const header = this.buildReportHeader(dateFormatted, locationName, stats.total);
  if (header) reportParts.push(header);
  
  // 3.2 游客统计
  const visitorSection = this.buildVisitorSection(stats.visitor);
  if (visitorSection) reportParts.push(visitorSection);
  
  // 3.3 员工统计
  const employeeSection = this.buildEmployeeSection(stats.employee);
  if (employeeSection) reportParts.push(employeeSection);
  
  // 3.4 出诊统计
  const outcallSection = this.buildOutcallSection(stats.outcall);
  if (outcallSection) reportParts.push(outcallSection);
  
  // 4. 拼接并返回
  const report = reportParts.join('\n');
  console.log('[formatDailyReport] 生成的日报:', report);
  
  return report;
}

// ==================== 构建报告标题 ====================
buildReportHeader(dateFormatted, locationName, total) {
  if (total === 0) {
    console.log('[buildReportHeader] 无记录，不显示标题');
    return '';
  }
  
  return `${dateFormatted}北京欢乐谷医务室（${locationName}）当日接诊${total}人。`;
}

// ==================== 构建游客统计 ====================
buildVisitorSection(visitorStats) {
  if (!visitorStats || visitorStats.length === 0) {
    console.log('[buildVisitorSection] 无游客记录');
    return '';
  }
  
  const visitorTotal = visitorStats.reduce((sum, item) => sum + item.total, 0);
  const visitorParts = visitorStats.map(item => {
    if (item.locations && item.locations.length > 0) {
      // 有地点的疾病：疾病X人（地点1X人，地点2X人）
      const locationParts = item.locations.map(loc => `${loc.name}${loc.count}人`);
      return `${item.disease}${item.total}人（${locationParts.join('，')}）`;
    }
    // 无地点的疾病：疾病X人
    return `${item.disease}${item.total}人`;
  });
  
  return `游客${visitorTotal}人：${visitorParts.join('，')}。`;
}

// ==================== 构建员工统计 ====================
buildEmployeeSection(employeeStats) {
  if (!employeeStats || employeeStats.length === 0) {
    console.log('[buildEmployeeSection] 无员工记录');
    return '';
  }
  
  const employeeTotal = employeeStats.reduce((sum, item) => sum + item.total, 0);
  const employeeParts = employeeStats.map(item => `${item.disease}${item.total}人`);
  
  return `员工${employeeTotal}人：${employeeParts.join('，')}。`;
}

// ==================== 构建出诊统计 ====================
buildOutcallSection(outcallStats) {
  if (!outcallStats || outcallStats.length === 0) {
    console.log('[buildOutcallSection] 无出诊记录');
    return '';
  }
  
  const outcallTotal = outcallStats.reduce((sum, item) => sum + item.count, 0);
  const outcallParts = outcallStats.map(item => `${item.location}${item.count}次`);
  
  return `出诊${outcallTotal}次：${outcallParts.join('，')}。`;
}

// ==================== 格式化日期 ====================
formatDate(dateStr) {
  const date = new Date(dateStr);
  const year = date.getFullYear();
  const month = date.getMonth() + 1;
  const day = date.getDate();
  return `${year}年${month}月${day}日`;
}

// ==================== 聚合统计数据 ====================
aggregateStats(records) {
  const stats = {
    total: records.length,
    visitor: [],
    employee: [],
    outcall: []
  };
  
  // 疾病统计
  const visitorDiseaseMap = new Map();
  const employeeDiseaseMap = new Map();
  const outcallLocationMap = new Map();
  
  records.forEach(record => {
    const identity = record.identity || '游客';
    const disease = record.diseaseName || record.diagnosis || '未知';
    const isOutcall = record.isOutcall || record.visitType === 'outcall';
    const location = record.injuryLocation || '';
    
    // 统计游客疾病
    if (identity === '游客') {
      if (!visitorDiseaseMap.has(disease)) {
        visitorDiseaseMap.set(disease, {
          disease,
          total: 0,
          locations: []
        });
      }
      const diseaseData = visitorDiseaseMap.get(disease);
      diseaseData.total++;
      
      // 统计地点（如果有）
      if (location) {
        const locIndex = diseaseData.locations.findIndex(l => l.name === location);
        if (locIndex >= 0) {
          diseaseData.locations[locIndex].count++;
        } else {
          diseaseData.locations.push({ name: location, count: 1 });
        }
      }
    }
    
    // 统计员工疾病
    if (identity === '员工') {
      if (!employeeDiseaseMap.has(disease)) {
        employeeDiseaseMap.set(disease, {
          disease,
          total: 0
        });
      }
      employeeDiseaseMap.get(disease).total++;
    }
    
    // 统计出诊
    if (isOutcall && location) {
      outcallLocationMap.set(location, (outcallLocationMap.get(location) || 0) + 1);
    }
  });
  
  // 转换为数组
  stats.visitor = Array.from(visitorDiseaseMap.values());
  stats.employee = Array.from(employeeDiseaseMap.values());
  stats.outcall = Array.from(outcallLocationMap.entries()).map(([location, count]) => ({
    location,
    count
  }));
  
  return stats;
}
```
